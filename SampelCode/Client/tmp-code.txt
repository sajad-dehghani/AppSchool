    private async Task UploadExcelAsync()
    {
        if (selectedFile == null)
        {
            snackBar.Add("لطفاً یک فایل انتخاب کنید!", Severity.Warning);
            return;
        }

        isProcessing = true;
        progressPercentage = 0;
        processedRows = 0;
        totalRows = 0;
        statusMessage = "در حال خواندن فایل اکسل...";
        StateHasChanged();

        try
        {
            var students = await ReadExcelAsync(selectedFile);

            if (students.Count == 0)
            {
                statusMessage = "فایل اکسل خالی است یا داده‌ای ندارد!";
                alertSeverity = Severity.Warning;
                return;
            }

            statusMessage = "در حال ارسال به سرور...";
            StateHasChanged();

            var response = await _httpClient.PostAsJsonAsync("api/Report_students/upload-excel", students);

            if (response.IsSuccessStatusCode)
            {
                statusMessage = $"اطلاعات با موفقیت ذخیره شد ({students.Count} رکورد)";
                alertSeverity = Severity.Success;
            }
            else
            {
                string errorMessage = await response.Content.ReadAsStringAsync();
                statusMessage = $"خطا در ارسال داده‌ها: {errorMessage}";
                alertSeverity = Severity.Error;
            }
        }
        catch (Exception ex)
        {
            statusMessage = $"خطایی رخ داد: {ex.Message}";
            alertSeverity = Severity.Error;
        }
        finally
        {
            isProcessing = false;
            StateHasChanged();
        }
    }
    private float SafeFloat(object? value, float defaultValue = 0f)
    {
        if (value == null) return defaultValue;
        var str = value.ToString()?.Trim();
        if (string.IsNullOrEmpty(str)) return defaultValue;
        return float.TryParse(str, out float result) ? result : defaultValue;
    }

    private long SafeLong(object? value, long defaultValue = 0)
    {
        if (value == null) return defaultValue;
        var str = value.ToString()?.Trim();
        if (string.IsNullOrEmpty(str)) return defaultValue;
        return long.TryParse(str, out long result) ? result : defaultValue;
    }

    private int SafeInt(object? value, int defaultValue = 0)
    {
        if (value == null) return defaultValue;
        var str = value.ToString()?.Trim();
        if (string.IsNullOrEmpty(str)) return defaultValue;
        return int.TryParse(str, out int result) ? result : defaultValue;
    }
